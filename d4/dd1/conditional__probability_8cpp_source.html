51double conditional_compute(double p_intersect, double p_given) {
52 if (p_given <= 0.0 || p_given > 1.0) {
53 throw std::invalid_argument("The given probability must be in (0, 1].");
54 }
55 if (p_intersect < 0.0 || p_intersect > p_given) {
56 throw std::invalid_argument(
57 "P(intersection) must be between 0 and the given probability.");
58 }
59
60 return p_intersect / p_given;
61}
